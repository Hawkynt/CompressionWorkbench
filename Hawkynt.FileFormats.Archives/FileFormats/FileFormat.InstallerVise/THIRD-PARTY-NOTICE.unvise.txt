FileFormat.InstallerVise is converted from unvise, the portable extractor for classic Mac OS
Installer VISE archives:

  https://github.com/timdoug/unvise
  revision 01b08a1455792fc6268127ad9f67f9bb5130d8da
  files: transport.c, io.c (resource map, AppleDouble), init.c, inflate.c, catalog.c, extract.c

ViseTransport.cs   <- transport.c, io.c (read_appledouble, read_sidecar_resource_fork)
ViseTables.cs      <- io.c (resource_find), init.c, extract.c (load_data0, load_table)
ViseDeflate.cs     <- inflate.c (DEFLATE is then decoded by this repository's own decoder, not zlib)
ViseCatalog.cs     <- catalog.c
ViseInstaller.cs   <- extract.c (extraction, payload layouts, run_installer)
