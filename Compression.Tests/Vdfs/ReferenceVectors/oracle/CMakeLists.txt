cmake_minimum_required(VERSION 3.16)
project(vdforacle CXX C)
set(CMAKE_CXX_STANDARD 20)
set(ZK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_ASAN OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_MMAP OFF CACHE BOOL "" FORCE)
set(ZK_ENABLE_ZIPPED_VDF ON CACHE BOOL "" FORCE)
# Point ZENKIT_DIR at a ZenKit checkout (commit e1c1044e4e0708b05db152ede44393f78d935ad5).
set(ZENKIT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ZenKit" CACHE PATH "ZenKit source")
add_subdirectory(${ZENKIT_DIR} zenkit)
add_executable(vdforacle vdforacle.cc)
target_link_libraries(vdforacle PRIVATE zenkit)
target_link_options(vdforacle PRIVATE -static)
